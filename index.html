Program exited successfully. " << endl; } /* * prompts for a date to be entered * and call create_datestamp() */ void string_function() { string date; cout << me << ", this is option 1" << endl; cout << "Please input a date (January 21): " << endl; getline(cin >> ws, date); // call create_datestamp create_datestamp(date); } /* * prompts for a number and calculate * the n-th Fibonacci number */ void number_function() { cout << me << ", this is option 2" << endl; int number = 0; int fn; int fn_1; int fn_2; cout << "Please input a number: " << endl; cin >> number; if (number >= 0) { for (int i = 0; i <= number; i++) { if (i == 0) { cout << 0; fn_2 = 0; } if (i == 1) { cout << 1; fn_1 = 1; } fn = fn_1 + fn_2; cout << fn; fn_2 = fn_1; fn_1 = fn; } cout << fn << endl; } else { cout << "Enter a positive whole number " << "greater than or equal to 0" << endl; } } // creates and prints the datestamp void create_datestamp(string datestr) { // first three characters of datestr string string1; // the position of white space int pos; // substring of all characters after the space string string2; // assign first three characters of datestr string1.assign(datestr, 0, 3); // find the position of white space pos = datestr.find(" "); /* * assign substring of datestr to string2 * starting at character after * the space to the end */ string2.assign(datestr, pos + 1, datestr.length() - pos - 1); // concatenate string1 and string2 datestr = string1 + string2; cout << "Datestamp created: " << datestr << endl; }
